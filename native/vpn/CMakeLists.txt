cmake_minimum_required(VERSION 3.25)
project(dsh_vpn LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "dsh-vpn requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OPENVPN3_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.cache/sources/openvpn3" CACHE PATH "Pinned OpenVPN3 source")
set(LWIP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.cache/sources/lwip" CACHE PATH "Pinned lwIP source")
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL REQUIRED)
find_package(asio CONFIG REQUIRED)
find_package(fmt CONFIG REQUIRED)
find_package(lz4 CONFIG REQUIRED)
find_package(jsoncpp CONFIG REQUIRED)
find_package(xxHash CONFIG REQUIRED)
include("${LWIP_DIR}/src/Filelists.cmake")
add_library(dsh_lwip STATIC ${lwipcore_SRCS} ${lwipcore4_SRCS})
target_include_directories(dsh_lwip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${LWIP_DIR}/src/include")
add_library(dsh_lwip_bridge STATIC src/lwip_stack.cpp)
target_link_libraries(dsh_lwip_bridge PUBLIC dsh_lwip ws2_32 bcrypt)
function(add_vpn_executable target source)
  add_executable(${target} "${source}" "${OPENVPN3_DIR}/openvpn/crypto/data_epoch.cpp")
  target_include_directories(${target} PRIVATE "${OPENVPN3_DIR}")
  target_compile_definitions(${target} PRIVATE USE_ASIO ASIO_STANDALONE USE_OPENSSL HAVE_LZ4 OPENVPN_EXTERNAL_TUN_FACTORY ASIO_DISABLE_LOCAL_SOCKETS TAP_WIN_COMPONENT_ID=tap0901 _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /bigobj)
  target_link_libraries(${target} PRIVATE dsh_lwip_bridge asio::asio fmt::fmt lz4::lz4 JsonCpp::JsonCpp xxHash::xxhash OpenSSL::SSL OpenSSL::Crypto ws2_32 crypt32 bcrypt iphlpapi fwpuclnt wininet setupapi advapi32 shell32 ole32 rpcrt4 wtsapi32)
endfunction()
add_vpn_executable(dsh-vpn src/main.cpp)

  add_executable(lwip-selftest tests/lwip_selftest.cpp)
